<script>
const SUPABASE_URL='https://zinqmmwprwvtjipfwfqp.supabase.co';
const SUPABASE_PUBLISHABLE_KEY='sb_publishable_TSOF9M-eL9DvKj42QKerxA_uZeWGw9h';
const sb=supabase.createClient(SUPABASE_URL,SUPABASE_PUBLISHABLE_KEY);
let globalEnabled=false, initializing=false;

const $=id=>document.getElementById(id);

async function login(){
  const {error}=await sb.auth.signInWithPassword({
    email:$('email').value.trim(),
    password:$('password').value
  });
  if(error){$('loginMsg').textContent=error.message;return}
  await init();
}

async function init(){
  if(initializing)return;
  initializing=true;
  try{
    const {data:{user}}=await sb.auth.getUser();
    if(!user){
      $('login').classList.remove('hidden');$('panel').classList.add('hidden');
      return;
    }
    $('login').classList.add('hidden');$('panel').classList.remove('hidden');
    await loadGlobal();
    await loadStreams();
  } finally { initializing=false; }
}

async function loadGlobal(){
  const {data,error}=await sb.from('app_config')
    .select('value').eq('key','streaming_enabled').maybeSingle();
  if(error){$('globalState').textContent=error.message;return}
  globalEnabled=!!data?.value;
  $('globalState').innerHTML=globalEnabled
    ? '<span class="ok">🟢 ACTIVADO</span>'
    : '<span class="off">🔴 DESACTIVADO</span>';
}

async function toggleGlobal(){
  const {error}=await sb.from('app_config')
    .upsert({key:'streaming_enabled',value:!globalEnabled});
  if(error)alert(error.message); else await loadGlobal();
}

async function addStream(){
  const payload={
    match_id:Number($('matchId').value),
    match_name:$('matchName').value.trim(),
    stream_url:$('url').value.trim(),
    referrer:$('referrer').value.trim()||null,
    user_agent:$('userAgent').value.trim()||null,
    starts_at:$('starts').value?new Date($('starts').value).toISOString():null,
    ends_at:$('ends').value?new Date($('ends').value).toISOString():null,
    enabled:false
  };
  if(!payload.match_id||!payload.match_name||!/^https?:\/\//i.test(payload.stream_url)){
    $('msg').textContent='Completa ID, nombre y una URL válida (http/https).';
    return;
  }
  const {error}=await sb.from('streams').insert(payload);
  $('msg').textContent=error
    ? error.message
    : 'Guardada como DESACTIVADA. Actívala después de verificar la autorización.';
  if(!error){
    ['matchId','matchName','url','referrer','userAgent','starts','ends'].forEach(id=>$(id).value='');
    await loadStreams();
  }
}

// Función para procesar y cargar la lista M3U en bloque
async function addBulkM3U(){
  const text = $('m3uInput').value.trim();
  const btn = $('btnM3u');
  const msg = $('m3uMsg');

  if(!text){
    msg.textContent = 'Por favor pega la lista M3U primero.';
    return;
  }

  btn.disabled = true;
  msg.textContent = 'Procesando lista M3U...';

  const lines = text.split('\n');
  const payloadList = [];
  let currentName = '';

  for (let i = 0; i < lines.length; i++) {
    const line = lines[i].trim();
    if (line.startsWith('#EXTINF:')) {
      const parts = line.split(',');
      if (parts.length > 1) {
        currentName = parts.slice(1).join(',').trim();
      }
    } else if (line.startsWith('http://') || line.startsWith('https://')) {
      if (currentName) {
        payloadList.push({
          match_id: Math.floor(100000 + Math.random() * 900000), // Genera un ID automático
          match_name: currentName,
          stream_url: line,
          enabled: true // Se agregan activadas por defecto
        });
        currentName = '';
      }
    }
  }

  if (payloadList.length === 0) {
    msg.textContent = 'No se encontraron canales válidos en el texto ingresado.';
    btn.disabled = false;
    return;
  }

  msg.textContent = `Insertando ${payloadList.length} canales en la base de datos...`;

  const { error } = await sb.from('streams').insert(payloadList);

  if (error) {
    msg.textContent = 'Error al guardar: ' + error.message;
  } else {
    msg.textContent = `¡Éxito! Se guardaron ${payloadList.length} canales correctamente.`;
    $('m3uInput').value = '';
    await loadStreams();
  }

  btn.disabled = false;
}

async function loadStreams(){
  const {data,error}=await sb.from('streams').select('*').order('created_at',{ascending:false});
  if(error){$('streams').textContent=error.message;return}$('streams').innerHTML=data.map(s=>`
    <div class="row">
      <div>
        <b>${esc(s.match_name)}</b>
        <div class="muted">
          ID ${s.match_id}<br>
          ${esc(s.stream_url)}<br>
          ${s.enabled?'<span class="ok">🟢 ACTIVA</span>':'<span class="off">🔴 DESACTIVADA</span>'}
        </div>
      </div>
      <div class="actions">
        <button class="secondary switch" onclick="setStream('${s.id}',${!s.enabled})">
          ${s.enabled?'Desactivar':'Activar'}
        </button>
        <button class="danger switch" onclick="removeStream('${s.id}')">Eliminar</button>
      </div>
    </div>`).join('') || '<p class="muted">No hay transmisiones.</p>';
}

async function setStream(id,enabled){
  const {error}=await sb.from('streams').update({enabled}).eq('id',id);
  if(error)alert(error.message); else await loadStreams();
}

async function removeStream(id){
  if(!confirm('¿Eliminar esta transmisión?'))return;
  const {error}=await sb.from('streams').delete().eq('id',id);
  if(error)alert(error.message); else await loadStreams();
}

async function logout(){
  await sb.auth.signOut();
  await init();
}

function esc(x){
  return String(x??'').replace(/[&<>"']/g,m=>({
    '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'
  }[m]));
}

sb.auth.onAuthStateChange(()=>init());
